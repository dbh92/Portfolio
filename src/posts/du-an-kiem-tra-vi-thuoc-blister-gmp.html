Trong nhà máy dược, trạm vision trên máy đóng vỉ (blister) kiểm từng hốc trước khi hàn màng nhôm: có viên chưa, viên có mẻ vỡ không, có lẫn viên của sản phẩm khác không, có đốm bẩn không. Hai điều làm bài toán này khác các trạm kiểm tra thông thường. Thứ nhất, màng PVC tạo vệt chói cong ngay sát mép viên, đúng chỗ cần đo mẻ. Thứ hai, ngành dược yêu cầu toàn vẹn dữ liệu: mọi kết quả, mọi lần đổi ngưỡng phải được ghi lại, ai làm, lúc nào, vì sao, và không ai sửa được mà không bị phát hiện. Dự án này làm cả hai phần.

Mục tiêu dự án(1) Kiểm 10 hốc mỗi vỉ: thiếu viên, mẻ, sai màu, đốm bẩn, bền với vệt chói và vỉ lệch. (2) Đo được mẻ nhỏ tới mức nào và chọn ngưỡng bằng dữ liệu. (3) Nhật ký chỉ ghi thêm, chuỗi băm phát hiện bản ghi bị sửa. (4) Phân quyền đổi công thức có lý do, và hiểu vì sao chuỗi băm thường chưa đủ, cần khóa bí mật.

2. Yêu cầu

  • Vỉ lệch ±10 pixel: định vị bằng hai lỗ định vị (lỗ đục trên màng), không giả định vị trí hốc.
  • Đèn mạnh yếu ±10%: chuẩn hóa độ sáng theo lỗ định vị.
  • Vệt chói trên vòm PVC không được làm viên tốt thành "mẻ" và không được che mẻ thật.
  • Trạm không thấy lỗ định vị thì báo lỗi đo, không phán OK/NG.
  • Nhật ký chỉ ghi thêm, có chuỗi băm; mọi lần đổi công thức (kể cả bị từ chối) đều được ghi.

3. Kiến thức sử dụng

  • Phân ngưỡng và contour, morphology (opening để bỏ vệt chói mảnh).
  • Fitting đường tròn và loại điểm lệch.
  • Kiểm tra màu: Lab, ΔE2000.
  • MES và truy xuất nguồn gốc.

4. Dữ liệu: vỉ mô phỏng

Viên trắng nằm trong hốc PVC trong, bên dưới là tấm đỡ tối (kiểm trước khi hàn màng nhôm, lúc tương phản viên/nền cao nhất). Mỗi hốc có một vệt chói cong ở vòm PVC, vị trí ngẫu nhiên, thường chạm mép viên. Viên lệch trong hốc ±4 pixel, đường kính dao động 1%. Mẻ được tạo bằng cách cắt đúng một tỉ lệ diện tích cho trước (giải phương trình diện tích hình viên phân), để đo được "bắt được mẻ từ bao nhiêu phần trăm".

import cv2
import numpy as np

PX_MM = 6.0
H, W = 330, 720
HANG, COT, BUOC = 2, 5, 132                    # 2 × 5 hốc, bước 22 mm
GOC_HOC = np.array([(96 + c * BUOC, 99 + r * BUOC) for r in range(HANG) for c in range(COT)], float)
LO_DANG_KY = np.array([[30.0, 30.0], [690.0, 30.0]])          # hai lỗ định vị trên màng (pixel, danh định)
R_VIEN, R_HOC = 30, 36                         # viên 10 mm, hốc 12 mm
MAU_VIEN = np.array([0.82, 0.84, 0.85], np.float32)          # viên trắng (B, G, R, tuyến tính)
MAU_LA = np.array([0.55, 0.80, 0.86], np.float32)            # viên của sản phẩm khác lẫn vào: ngà vàng


def tao_vi(rng, ty_le_loi=0.35):
    """Mỗi vỉ có xác suất ty_le_loi chứa một hốc lỗi: thiếu viên, mẻ (mất 4–25% diện tích), sai màu, đốm bẩn."""
    loi = {}
    if rng.random() < ty_le_loi:
        loi[int(rng.integers(0, HANG * COT))] = ["thiếu", "mẻ", "sai màu", "đốm bẩn"][rng.integers(0, 4)]
    return loi


def ve_vien(anh, tam, r, mau, mat_manh=0.0, goc_me=0.0):
    lop = np.zeros(anh.shape[:2], np.uint8)
    cv2.circle(lop, tuple(np.round(tam).astype(int)), int(round(r)), 255, -1, cv2.LINE_AA)
    if mat_manh > 0:                                             # cắt một mảnh hình viên phân: mất đúng mat_manh diện tích
        lo, hi = -r, r                                           # tìm khoảng cách dây cung d bằng chia đôi
        for _ in range(50):
            d = (lo + hi) / 2
            phan = (r * r * np.arccos(d / r) - d * np.sqrt(r * r - d * d)) / (np.pi * r * r)
            lo, hi = (d, hi) if phan > mat_manh else (lo, d)
        huong = np.array([np.cos(goc_me), np.sin(goc_me)])
        yy, xx = np.mgrid[:anh.shape[0], :anh.shape[1]]
        lop[((xx - tam[0]) * huong[0] + (yy - tam[1]) * huong[1]) > d] = 0
    m = lop.astype(np.float32)[..., None] / 255
    yy, xx = np.mgrid[:anh.shape[0], :anh.shape[1]]
    bong = 1 - 0.12 * (((xx - tam[0]) ** 2 + (yy - tam[1]) ** 2) / r ** 2)   # viên lồi: tối dần ra mép
    anh[:] = anh * (1 - m) + (mau * np.clip(bong, 0.8, 1)[..., None]) * m


def chup_vi(loi, rng, danh_dinh=False, ty_le_me=None):
    anh = np.full((H, W, 3), 0.08, np.float32)                   # tấm đỡ tối dưới màng PVC trong
    lech = np.zeros(2) if danh_dinh else rng.uniform(-10, 10, 2)
    for p in LO_DANG_KY + lech:
        cv2.circle(anh, tuple(np.round(p).astype(int)), 8, (0.95, 0.95, 0.95), -1, cv2.LINE_AA)
    for k, g in enumerate(GOC_HOC + lech):
        cv2.circle(anh, tuple(np.round(g).astype(int)), R_HOC, (0.13, 0.13, 0.13), -1, cv2.LINE_AA)   # đáy hốc
        kieu = loi.get(k)
        if kieu != "thiếu":
            tam = g + (0 if danh_dinh else rng.uniform(-4, 4, 2))
            r = R_VIEN * (1 if danh_dinh else rng.normal(1, 0.01))
            mau = MAU_LA if kieu == "sai màu" else MAU_VIEN
            mat = 0.0
            if kieu == "mẻ":
                mat = ty_le_me if ty_le_me is not None else rng.uniform(0.04, 0.25)
            ve_vien(anh, tam, r, mau, mat, rng.uniform(0, 2 * np.pi))
            if kieu == "đốm bẩn":
                d = tam + rng.uniform(-15, 15, 2)
                cv2.circle(anh, tuple(np.round(d).astype(int)), int(rng.integers(2, 4)), (0.25, 0.22, 0.20), -1)
        goc = rng.uniform(0, 2 * np.pi) if not danh_dinh else 0.8
        cv2.ellipse(anh, tuple(np.round(g).astype(int)), (R_HOC - 3, R_HOC - 3), np.degrees(goc), 0, 70,
                    (0.97, 0.97, 0.97), 3, cv2.LINE_AA)        # vệt chói cong trên vòm PVC
    anh *= 1.0 if danh_dinh else rng.uniform(0.9, 1.1)
    return np.clip(anh + rng.normal(0, 0.01, anh.shape), 0, 1).astype(np.float32)

5. Lời giải 1: diện tích sáng trong ô cố định

Ô cố định quanh mỗi hốc, đếm pixel sáng hơn 0,5, so với diện tích dạy, cho phép ±10%.

# Lời giải 1: ô hốc cố định, ngưỡng sáng, đếm diện tích, so với diện tích dạy (±10%)
def dien_tich_1(anh, k):
    x, y = GOC_HOC[k].astype(int)
    o = anh[y - R_HOC:y + R_HOC, x - R_HOC:x + R_HOC].mean(axis=2)
    return int((o > 0.5).sum())


rng_day = np.random.default_rng(1)
DT_DAY = np.mean([dien_tich_1(chup_vi({}, rng_day, danh_dinh=True), k) for k in range(HANG * COT)])


def kiem_vi_1(anh):
    return {k: ("OK" if abs(dien_tich_1(anh, k) / DT_DAY - 1) <= 0.10 else "NG") for k in range(HANG * COT)}

6. Lời giải 2: định vị, fit đường tròn bền, đo trong đường tròn

  1. Định vị và chuẩn hóa sáng bằng hai lỗ định vị trắng.
  2. Mặt nạ viên: ngưỡng trên ảnh đã chuẩn hóa, opening 9 × 9 bỏ phần vệt chói mảnh, lấy blob lớn nhất, lấp lỗ (để đốm tối trên viên vẫn thuộc viên).
  3. Fit đường tròn bền trên biên viên: bình phương tối thiểu, lặp lại sau khi bỏ điểm lệch quá 2 pixel. Mép bị mẻ nằm trong đường tròn, vệt chói dính mép nằm ngoài: cả hai đều bị loại khỏi phép fit.
  4. Độ đầy = phần diện tích viên nằm trong đường tròn fit (bán kính trừ 1,5 pixel để bỏ mép răng cưa). Vệt chói nằm ngoài đường tròn nên không làm sai; mẻ làm thiếu một mảnh bên trong nên độ đầy giảm đúng bằng tỉ lệ mẻ.
  5. Màu: Lab của phần lõi viên, ΔE00 so với màu dạy; đốm: số pixel tối hơn trung vị lõi viên 0,2.
def de2000(lab1, lab2):
    """CIEDE2000 (bài kiểm tra màu: Lab, ΔE2000)."""
    L1, a1, b1 = np.moveaxis(np.asarray(lab1, float), -1, 0)
    L2, a2, b2 = np.moveaxis(np.asarray(lab2, float), -1, 0)
    C_tb = (np.hypot(a1, b1) + np.hypot(a2, b2)) / 2
    G = 0.5 * (1 - np.sqrt(C_tb ** 7 / (C_tb ** 7 + 25.0 ** 7)))
    a1p, a2p = (1 + G) * a1, (1 + G) * a2
    C1p, C2p = np.hypot(a1p, b1), np.hypot(a2p, b2)
    h1p, h2p = np.degrees(np.arctan2(b1, a1p)) % 360, np.degrees(np.arctan2(b2, a2p)) % 360
    dh = h2p - h1p
    dh = np.where(dh > 180, dh - 360, np.where(dh < -180, dh + 360, dh))
    dh = np.where(C1p * C2p == 0, 0, dh)
    dL, dC = L2 - L1, C2p - C1p
    dH = 2 * np.sqrt(C1p * C2p) * np.sin(np.radians(dh / 2))
    L_tb, Cp_tb, tong_h = (L1 + L2) / 2, (C1p + C2p) / 2, h1p + h2p
    h_tb = np.where(C1p * C2p == 0, tong_h, np.where(np.abs(h1p - h2p) <= 180, tong_h / 2,
                    np.where(tong_h < 360, (tong_h + 360) / 2, (tong_h - 360) / 2)))
    T = (1 - 0.17 * np.cos(np.radians(h_tb - 30)) + 0.24 * np.cos(np.radians(2 * h_tb))
         + 0.32 * np.cos(np.radians(3 * h_tb + 6)) - 0.20 * np.cos(np.radians(4 * h_tb - 63)))
    SL = 1 + 0.015 * (L_tb - 50) ** 2 / np.sqrt(20 + (L_tb - 50) ** 2)
    SC, SH = 1 + 0.045 * Cp_tb, 1 + 0.015 * Cp_tb * T
    RT = -np.sin(np.radians(60 * np.exp(-((h_tb - 275) / 25) ** 2))) * 2 * np.sqrt(Cp_tb ** 7 / (Cp_tb ** 7 + 25.0 ** 7))
    return np.sqrt((dL / SL) ** 2 + (dC / SC) ** 2 + (dH / SH) ** 2 + RT * (dC / SC) * (dH / SH))


def can_vi(anh):
    """Độ lệch của màng theo hai lỗ định vị, và hệ số độ sáng (lỗ định vị trắng 0,95 khi đèn danh định)."""
    sang = (anh.mean(axis=2) > 0.6).astype(np.uint8)
    n, _, st, tam = cv2.connectedComponentsWithStats(sang)
    lech, do_sang = [], []
    for p in LO_DANG_KY:
        uv = [k for k in range(1, n) if 120 < st[k, cv2.CC_STAT_AREA] < 300 and np.linalg.norm(tam[k] - p) < 25]
        if len(uv) != 1:
            return None, None
        lech.append(tam[uv[0]] - p)
        x, y = np.round(tam[uv[0]]).astype(int)
        do_sang.append(anh[y - 3:y + 4, x - 3:x + 4].mean() / 0.95)
    return np.mean(lech, axis=0), float(np.mean(do_sang))


def fit_duong_tron(diem, lap=4, nguong=2.0):
    """Fit đường tròn bình phương tối thiểu (Kåsa), lặp lại sau khi bỏ điểm lệch > nguong pixel
    (mép bị mẻ nằm trong, vệt chói dính mép nằm ngoài: cả hai đều là điểm lệch)."""
    giu = np.ones(len(diem), bool)
    for _ in range(lap):
        x, y = diem[giu, 0], diem[giu, 1]
        A = np.column_stack([x, y, np.ones_like(x)])
        a, b, c = np.linalg.lstsq(A, x * x + y * y, rcond=None)[0]
        cx, cy = a / 2, b / 2
        r = np.sqrt(c + cx * cx + cy * cy)
        giu = np.abs(np.hypot(diem[:, 0] - cx, diem[:, 1] - cy) - r) < nguong
    return cx, cy, r


def dac_trung_vien(anh, tam_hoc, do_sang):
    """None nếu hốc trống; ngược lại: độ đầy trong đường tròn fit, bán kính, Lab, số pixel đốm."""
    x, y = np.round(tam_hoc).astype(int)
    o = anh[y - R_HOC - 4:y + R_HOC + 4, x - R_HOC - 4:x + R_HOC + 4]
    xam = o.mean(axis=2) / do_sang
    mat_na = (xam > 0.45).astype(np.uint8)
    mat_na = cv2.morphologyEx(mat_na, cv2.MORPH_OPEN, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (9, 9)))
    n, nhan, st, _ = cv2.connectedComponentsWithStats(mat_na)
    if n < 2 or st[1:, cv2.CC_STAT_AREA].max() < 0.3 * np.pi * R_VIEN ** 2:
        return None
    k = 1 + np.argmax(st[1:, cv2.CC_STAT_AREA])
    vien_bien, _ = cv2.findContours((nhan == k).astype(np.uint8), cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)
    bien = max(vien_bien, key=len)
    vien = np.zeros_like(mat_na)
    cv2.drawContours(vien, [bien], -1, 1, -1)                          # lấp lỗ: đốm tối trên viên vẫn thuộc viên
    cx, cy, r = fit_duong_tron(bien[:, 0, :].astype(float))
    dia = np.zeros_like(vien)
    cv2.circle(dia, (int(round(cx)), int(round(cy))), int(round(r - 1.5)), 1, -1)
    trong = cv2.erode(vien, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (11, 11))) > 0
    lab = cv2.cvtColor((np.median(o[trong], axis=0) / do_sang).astype(np.float32)[None, None], cv2.COLOR_LBGR2Lab)[0, 0]
    trung_vi = np.median(xam[trong])
    return {"do_day": float((vien & dia).sum() / dia.sum()), "ban_kinh": float(r / R_VIEN),
            "lab": lab, "dom": int(np.sum(xam[trong] < trung_vi - 0.2))}

def kiem_hoc(f, nguong_day=0.97, nguong_de=4.0):
    if f is None:
        return "thiếu"
    if f["do_day"] < nguong_day:
        return "mẻ"
    if de2000(f["lab"], LAB_DAY) > nguong_de:
        return "sai màu"
    if f["dom"] >= 6:
        return "đốm bẩn"
    return "OK"

8. Toàn vẹn dữ liệu: nhật ký chuỗi băm và phân quyền

Yêu cầu về hồ sơ điện tử trong ngành dược (FDA 21 CFR Part 11, EU GMP Annex 11) xoay quanh vài nguyên tắc dễ hiểu: hồ sơ phải ghi được ai, làm gì, lúc nào; thay đổi phải có lý do và không xóa dấu vết cũ; chỉ người có quyền mới được thay đổi; và mọi sửa đổi trái phép phải phát hiện được. Đoạn code dưới hiện thực các nguyên tắc đó ở mức trạm: mỗi bản ghi chứa mã băm SHA-256 của bản ghi trước, nên sửa một bản ghi bất kỳ làm đứt chuỗi từ chỗ đó.

import hashlib
import json


class NhatKy:
    """Nhật ký chỉ ghi thêm (append-only), mỗi bản ghi chứa mã băm của bản ghi trước: sửa hay xóa một
    bản ghi bất kỳ làm đứt chuỗi từ chỗ đó trở đi."""

    def __init__(self):
        self.ban_ghi = []

    def ghi(self, thoi_gian, nguoi, loai, noi_dung):
        truoc = self.ban_ghi[-1]["bam"] if self.ban_ghi else "0" * 64
        bg = {"stt": len(self.ban_ghi), "thoi_gian": thoi_gian, "nguoi": nguoi, "loai": loai,
              "noi_dung": noi_dung, "bam_truoc": truoc}
        bg["bam"] = hashlib.sha256(json.dumps(bg, ensure_ascii=False, sort_keys=True).encode()).hexdigest()
        self.ban_ghi.append(bg)

    def kiem_tra(self):
        """Trả về chỉ số bản ghi đầu tiên bị sửa, hoặc None nếu chuỗi nguyên vẹn."""
        truoc = "0" * 64
        for bg in self.ban_ghi:
            than = {k: v for k, v in bg.items() if k != "bam"}
            if bg["bam_truoc"] != truoc or hashlib.sha256(
                    json.dumps(than, ensure_ascii=False, sort_keys=True).encode()).hexdigest() != bg["bam"]:
                return bg["stt"]
            truoc = bg["bam"]
        return None

class NhatKyHMAC(NhatKy):
    """Như NhatKy nhưng băm bằng HMAC với khóa bí mật chỉ trạm có (lưu trong TPM/HSM hoặc kho khóa của hệ thống)."""

    def __init__(self, khoa):
        super().__init__()
        self.khoa = khoa

    def _bam(self, than):
        return hmac.new(self.khoa, json.dumps(than, ensure_ascii=False, sort_keys=True).encode(), "sha256").hexdigest()

    def ghi(self, thoi_gian, nguoi, loai, noi_dung):
        truoc = self.ban_ghi[-1]["bam"] if self.ban_ghi else "0" * 64
        bg = {"stt": len(self.ban_ghi), "thoi_gian": thoi_gian, "nguoi": nguoi, "loai": loai,
              "noi_dung": noi_dung, "bam_truoc": truoc}
        bg["bam"] = self._bam(bg)
        self.ban_ghi.append(bg)

    def kiem_tra(self):
        truoc = "0" * 64
        for bg in self.ban_ghi:
            than = {k: v for k, v in bg.items() if k != "bam"}
            if bg["bam_truoc"] != truoc or not hmac.compare_digest(self._bam(than), bg["bam"]):
                return bg["stt"]
            truoc = bg["bam"]
        return None

Chuỗi SHA-256 thường không có gì bí mật: ai có file đều tính lại được, nên chuỗi mới vẫn "nguyên vẹn". Chuỗi băm chỉ phát hiện sửa đổi khi kẻ sửa không thể tạo lại mã băm hợp lệ. HMAC dùng một khóa bí mật chỉ trạm có (lưu trong TPM, HSM hoặc kho khóa của hệ điều hành, không nằm trong mã nguồn): kẻ sửa không có khóa thì mọi mã băm tính lại đều sai, ngay từ bản ghi đầu tiên. Cách bổ sung rẻ và hiệu quả: định kỳ gửi mã băm của bản ghi mới nhất sang một hệ thống khác (MES, máy chủ QA). Muốn sửa lén thì phải sửa được cả ở hai nơi.
