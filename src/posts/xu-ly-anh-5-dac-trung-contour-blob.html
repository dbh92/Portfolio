Bài phân ngưỡng và contour đã dạy bạn đếm vật. Nhưng trạm kiểm tra thật cần nhiều hơn: vật này tròn hay méo, đặt thẳng hay lệch góc, có lỗ ở giữa không, có bị sứt mẻ không. Những câu hỏi đó trả lời bằng đặc trưng hình học của từng contour. Kỹ thuật này gọi là blob analysis và là công cụ chính của Machine Vision cổ điển: nhanh, dễ giải thích, không cần dữ liệu huấn luyện.

Mục tiêu cuối bàiViết được hàm nhận mặt nạ nhị phân và trả về bảng đặc trưng của từng vật (diện tích, chu vi, tâm, độ tròn, độ đặc, góc xoay), rồi phân loại vật theo bảng đó.

1. Ảnh mẫu

Tạo một ảnh nhị phân có đủ các tình huống: tròn, vuông, chữ nhật xoay, tam giác, vòng đệm có lỗ và một vật bị cắt ở mép ảnh.

5. Góc xoay với minAreaRect: cách dùng an toàn

cv2.minAreaRect trả về ((cx, cy), (w, h), góc): hình chữ nhật nhỏ nhất bao vật, kể cả khi vật xoay. Vấn đề: quy ước góc đã thay đổi giữa các phiên bản OpenCV, và w, h có thể bị hoán đổi. Với chữ nhật xoay 30° ở trên, OpenCV bạn đang dùng có thể trả về 30, −60 hoặc 60 tùy phiên bản. Đừng dùng thẳng giá trị góc. Hãy tính góc của cạnh dài từ 4 đỉnh, kết quả sẽ giống nhau trên mọi phiên bản:

import math

import cv2
import numpy as np


def khung_xoay(c):
    """Trả về (dài, rộng, góc cạnh dài trong [0, 180)), không phụ thuộc phiên bản OpenCV."""
    rect = cv2.minAreaRect(c)
    p = cv2.boxPoints(rect)                      # 4 đỉnh theo thứ tự liền kề
    c1, c2 = p[1] - p[0], p[2] - p[1]            # hai cạnh kề nhau
    canh_dai = c1 if np.linalg.norm(c1) >= np.linalg.norm(c2) else c2
    goc = math.degrees(math.atan2(canh_dai[1], canh_dai[0])) % 180
    dai, rong = max(rect[1]), min(rect[1])
    return dai, rong, goc


img = cv2.imread("vat_the.png", cv2.IMREAD_GRAYSCALE)
cs, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

    print(f"khung thẳng {w:3d}×{h:3d} | khung xoay {dai:5.1f}×{rong:5.1f} góc {goc:5.1f}°")

6. Bảng đặc trưng và phân loại vật

Gom tất cả thành một hàm trả về danh sách dict, mỗi dict là một vật. Đây là cấu trúc dữ liệu bạn sẽ dùng lại ở dự án chặng 3 và chặng 7:

import math

import cv2
import numpy as np


def khung_xoay(c):
    rect = cv2.minAreaRect(c)
    p = cv2.boxPoints(rect)
    c1, c2 = p[1] - p[0], p[2] - p[1]
    canh_dai = c1 if np.linalg.norm(c1) >= np.linalg.norm(c2) else c2
    return max(rect[1]), min(rect[1]), math.degrees(math.atan2(canh_dai[1], canh_dai[0])) % 180


def phan_tich_blob(mask, dien_tich_min=200, cao_hang=180):
    """mask: vật trắng nền đen. Trả về danh sách đặc trưng, xếp theo hàng (cao cao_hang px) rồi trái → phải."""
    cao, rong_anh = mask.shape[:2]
    cs, h = cv2.findContours(mask, cv2.RETR_CCOMP, cv2.CHAIN_APPROX_SIMPLE)
    if h is None:
        return []
    h = h[0]
    ket_qua = []
    for i, c in enumerate(cs):
        if h[i][3] != -1:
            continue
        A = cv2.contourArea(c)
        if A < dien_tich_min:
            continue
        P = cv2.arcLength(c, True)
        x, y, w, hh = cv2.boundingRect(c)
        M = cv2.moments(c)
        dai, rong, goc = khung_xoay(c)
        lo = [j for j in range(len(cs)) if h[j][3] == i]
        ket_qua.append({
            "tam": (round(M["m10"] / M["m00"], 1), round(M["m01"] / M["m00"], 1)),
            "dien_tich": round(A - sum(cv2.contourArea(cs[j]) for j in lo)),
            "do_tron": round(4 * math.pi * A / P ** 2, 3),
            "do_dac": round(A / cv2.contourArea(cv2.convexHull(c)), 3),
            "so_dinh": len(cv2.approxPolyDP(c, 0.02 * P, True)),
            "dai": round(dai, 1), "rong": round(rong, 1), "goc": round(goc, 1),
            "so_lo": len(lo),
            "cham_mep": x == 0 or y == 0 or x + w == rong_anh or y + hh == cao,
        })
    ket_qua.sort(key=lambda v: (int(v["tam"][1] // cao_hang), v["tam"][0]))   # hàng trước, cột sau
    return ket_qua


def phan_loai(v):
    if v["cham_mep"]:
        return "bỏ qua (chạm mép)"
    if v["so_lo"] > 0:
        return "vòng đệm"
    if v["do_tron"] > 0.85:
        return "tròn"
    if v["so_dinh"] == 3:
        return "tam giác"
    if v["so_dinh"] == 4:
        return "vuông" if v["dai"] / v["rong"] < 1.15 else "chữ nhật"
    return "không xác định"


mask = cv2.imread("vat_the.png", cv2.IMREAD_GRAYSCALE)
for v in phan_tich_blob(mask):
    print(f"{phan_loai(v):18s} tâm {v['tam']}  S={v['dien_tich']}  tròn={v['do_tron']}  "
          f"{v['dai']}×{v['rong']} góc {v['goc']}")

Những quyết định thiết kế đáng học:

  • Loại vật chạm mép: vật bị cắt cho số đo sai (diện tích thiếu, độ tròn sai). Trên băng tải, vật đó sẽ được kiểm tra đầy đủ ở khung hình sau.
  • Thứ tự kiểm tra trong phan_loai: kiểm tra lỗ trước độ tròn, vì vòng đệm cũng tròn. Kiểm tra độ tròn trước số đỉnh, vì approxPolyDP biến hình tròn thành đa giác 8 đỉnh.
  • Sắp xếp theo hàng rồi theo cột: chia ảnh thành các dải ngang cao cao_hang pixel (theo bố trí khay, ở đây 2 hàng, mỗi hàng 180 px). Nếu chỉ sắp theo y, vòng đệm (tâm y = 270) sẽ đứng trước tam giác (y = 286,7) dù nằm bên phải, vì tâm tam giác thấp hơn. Lỗi này dễ làm sai số thứ tự vị trí khi báo NG.
  • approxPolyDP(c, 0.02*P): sai số cho phép là 2% chu vi. Tăng lên nếu cạnh vật bị răng cưa (ra quá nhiều đỉnh), giảm nếu góc bo tròn (ra quá ít đỉnh).

8. Từ đặc trưng đến quyết định OK/NG

Trong nhà máy, bảng đặc trưng được so với khoảng dung sai lấy từ mẫu chuẩn. Ví dụ quy tắc cho vòng đệm:

DUNG_SAI_VONG_DEM = {
    "dien_tich": (6200, 6800),    # pixel², từ đo 30 mẫu OK: trung bình ± 4 độ lệch chuẩn
    "do_tron": (0.85, 1.0),       # méo → NG
    "do_dac": (0.97, 1.0),        # sứt mép → độ đặc giảm
    "so_lo": (1, 1),              # thiếu hoặc thừa lỗ → NG
}


def kiem_tra(v, dung_sai):
    loi = [f"{k}={v[k]} ngoài [{lo}, {hi}]" for k, (lo, hi) in dung_sai.items()
           if not lo <= v[k] <= hi]
    return len(loi) == 0, loi


vong_dem = {"dien_tich": 6390, "do_tron": 0.891, "do_dac": 0.989, "so_lo": 1}
sut_me = {"dien_tich": 6010, "do_tron": 0.80, "do_dac": 0.93, "so_lo": 1}
print(kiem_tra(vong_dem, DUNG_SAI_VONG_DEM))
print(kiem_tra(sut_me, DUNG_SAI_VONG_DEM))

9. Bài tập

Bài 1. Tạo ảnh mới có hai hình tròn bán kính 40: một nguyên vẹn, một bị khoét "vết sứt" ở mép bằng hình tròn đen bán kính 15. So sánh độ tròn và độ đặc. Đặc trưng nào phát hiện sứt mẻ tốt hơn?

Cả hai đều giảm, nhưng độ đặc là chỉ số đặc hiệu cho sứt mẻ vì nó đo trực tiếp phần "khuyết" so với bao lồi, còn độ tròn giảm cả khi vật chỉ hơi bầu dục (không có lỗi khuyết). Bài 2. Viết hàm ve_bang(anh_mau, ds) vẽ lên ảnh màu: khung xoay (cv2.boxPoints + cv2.polylines), số thứ tự và tên loại tại tâm mỗi vật. Dùng kết quả phan_tich_blob và phan_loai ở mục 6.

Xem đáp án 
import cv2
import numpy as np


def ve_bang(anh_mau, ds_contour, ds_ten):
    ve = anh_mau.copy()
    for i, (c, ten) in enumerate(zip(ds_contour, ds_ten), 1):
        hop = cv2.boxPoints(cv2.minAreaRect(c)).astype(np.int32)
        mau = (0, 0, 255) if "bỏ qua" in ten else (0, 200, 0)
        cv2.polylines(ve, [hop], True, mau, 2)
        M = cv2.moments(c)
        cx, cy = int(M["m10"] / M["m00"]), int(M["m01"] / M["m00"])
        cv2.putText(ve, str(i), (cx - 6, cy + 6), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 0, 0), 2)
    return ve


mask = cv2.imread("vat_the.png", cv2.IMREAD_GRAYSCALE)
cs, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
ket_qua = ve_bang(cv2.cvtColor(mask, cv2.COLOR_GRAY2BGR), cs, ["vật"] * len(cs))
cv2.imwrite("bang_vat.png", ket_qua)
print("Đã lưu bang_vat.png")

In tên loại tiếng Việt lên ảnh: xem lưu ý về putText và font ở bài vẽ kết quả (chặng 2). Ở đây dùng số thứ tự và in bảng tên ra terminal là cách an toàn trên mọi phiên bản.

Bài 3. Một khay có 12 viên thuốc. Ảnh NG thiếu 1 viên, ảnh NG khác có 1 viên vỡ đôi. Chỉ dùng đặc trưng trong bài, bạn thiết kế quy tắc OK/NG thế nào để bắt được cả hai lỗi?

Xem đáp án
  • Thiếu viên: đếm vật hợp lệ (diện tích trong khoảng dung sai của một viên) khác 12 → NG.
  • Viên vỡ đôi: hai mảnh có diện tích khoảng một nửa viên chuẩn, thấp hơn ngưỡng dưới → mỗi mảnh là một vật "ngoài dung sai" → NG. Nếu hai mảnh vẫn dính nhau, độ đặc giảm và độ tròn giảm → NG.
  • Quy tắc tổng: OK khi và chỉ khi có đúng 12 vật, và mọi vật nằm trong dung sai diện tích, độ tròn, độ đặc. Báo NG kèm số thứ tự vị trí trên khay, nhờ sắp xếp theo hàng/cột.

Tóm tắt

  • Dùng RETR_CCOMP khi cần biết lỗ; h[i][3] là chỉ số cha.
  • Độ tròn của hình tròn số hóa khoảng 0,89; ngưỡng thực tế khoảng 0,85. Độ đặc phát hiện sứt mẻ.
  • Đừng tin trực tiếp góc của minAreaRect: tính góc cạnh dài từ boxPoints.
  • Loại vật chạm mép ảnh; tự sắp xếp vật theo hàng/cột.
  • contourArea và connectedComponentsWithStats cho diện tích khác nhau, chọn một và dùng nhất quán.
  • Quyết định OK/NG = so đặc trưng với khoảng dung sai, luôn trả về lý do.

Bài tiếp theo xử lý trường hợp khó mà contour thường thất bại: các vật dính nhau, tách bằng watershed.
