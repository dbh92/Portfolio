6.1. Mã khởi đầu: bề mặt xước sợi tổng hợp, chia theo lô

Vân xước sợi được tạo bằng nhiễu làm mờ rất lệch (σ 4 px theo một chiều, 0,5 px theo chiều kia) rồi xoay theo hướng vân của lô. Mỗi lô có độ sáng, độ thô của vân, hướng vân và độ dốc chiếu sáng riêng: đó là lý do phải chia theo lô. Nếu trộn ngẫu nhiên, ô của cùng một lô nằm ở cả train và test, mô hình được "xem trước" lô test và con số đánh giá sẽ đẹp hơn thực tế. Tập dữ liệu được làm giàu lỗi (35%) để có đủ mẫu NG; dây chuyền thật chỉ khoảng 3%.

# ===== du_lieu.py — bề mặt nhôm xước sợi tổng hợp, chia theo lô, có nhãn và loại lỗi =====
import math

import cv2
import numpy as np

S = 48                                  # kích thước một ô ảnh (px)


def thong_so_lo(lo):
    """Mỗi lô vật liệu có độ sáng, độ thô sợi, hướng sợi và độ dốc chiếu sáng hơi khác nhau."""
    r = np.random.default_rng(1000 + lo)
    return {"sang": r.uniform(125, 150), "tho": r.uniform(5, 8), "huong": r.uniform(-10, 10),
            "doc": r.uniform(-12, 12, 2)}


def ve_o(rng, p, loi=None, do_sang=1.0):
    """Một ô 48 × 48: vân xước sợi + độ dốc sáng + nhiễu; loi ∈ {None, 'xuoc', 'dom'}."""
    B = S + 24
    van = cv2.GaussianBlur(rng.normal(0, 1, (B, B)).astype(np.float32), (0, 0), sigmaX=4.0, sigmaY=0.5)
    M = cv2.getRotationMatrix2D((B / 2, B / 2), p["huong"] + rng.uniform(-3, 3), 1.0)
    van = cv2.warpAffine(van, M, (B, B))[12:12 + S, 12:12 + S]
    van /= van.std() + 1e-6
    yy, xx = np.mgrid[0:S, 0:S] / S - 0.5
    anh = p["sang"] + p["tho"] * van + p["doc"][0] * xx + p["doc"][1] * yy + rng.normal(0, 2.5, (S, S))
    if loi is not None:
        lop = np.zeros((S, S), np.float32)
        if loi == "xuoc":                              # vết xước mảnh, dài 12–30 px, hướng bất kỳ
            a, L = rng.uniform(0, math.pi), rng.uniform(12, 30)
            cx, cy = rng.uniform(12, 36, 2)
            d = (L / 2 * math.cos(a), L / 2 * math.sin(a))
            cv2.line(lop, (int(cx - d[0]), int(cy - d[1])), (int(cx + d[0]), int(cy + d[1])), 1.0, 1, cv2.LINE_AA)
            anh -= rng.uniform(28, 50) * lop
        else:                                          # đốm (lõm, bụi kẹt) bán kính 2–3 px
            c = tuple(int(v) for v in rng.uniform(6, 42, 2))
            cv2.circle(lop, c, int(rng.integers(2, 4)), 1.0, -1, cv2.LINE_AA)
            anh -= rng.uniform(30, 55) * cv2.GaussianBlur(lop, (0, 0), 0.7)
    return np.clip(anh * do_sang, 0, 255).astype(np.uint8)


def tao_lo(lo, so_o, ti_le_loi, do_sang=1.0, seed=None):
    rng = np.random.default_rng(lo if seed is None else seed)
    p = thong_so_lo(lo)
    X, y, loai = [], [], []
    for _ in range(so_o):
        loi = None
        if rng.random() < ti_le_loi:
            loi = "xuoc" if rng.random() < 0.6 else "dom"
        X.append(ve_o(rng, p, loi, do_sang))
        y.append(int(loi is not None))
        loai.append(loi or "ok")
    return np.array(X), np.array(y), np.array(loai)


if __name__ == "__main__":
    # 10 lô × 150 ô, làm giàu lỗi 35% (dây chuyền thật chỉ vài %: xem mục 7)

Phần 1: huấn luyện. Mạng 3 khối conv + BatchNorm, kết thúc bằng max pooling toàn ảnh: một vết lỗi nhỏ ở bất kỳ đâu cũng kích hoạt đầu ra, trong khi average pooling sẽ "pha loãng" nó với phần còn lại của ô. Tiền xử lý trừ trung bình từng ảnh (bỏ khác biệt độ sáng giữa các lô) nhưng chia cho một hằng số thay vì độ lệch chuẩn của ảnh (giữ thông tin tương phản của vết lỗi).

# ===== huan_luyen.py — CNN nhỏ phân loại ô OK/NG, chia train/val/test THEO LÔ =====
import time

import numpy as np
import torch
import torch.nn as nn

torch.manual_seed(0)
torch.set_num_threads(1)        # mô hình rất nhỏ: 1 luồng thường nhanh hơn nhiều luồng (xem chú thích bên dưới)

d = np.load("du_lieu.npz")
X, y, lo = d["X"], d["y"], d["lo"]
TAP = {"train": [1, 2, 3, 4, 5, 6], "val": [7, 8], "test": [9, 10]}
chia = {k: np.isin(lo, v) for k, v in TAP.items()}


class MangNho(nn.Module):
    """3 khối conv; gộp bằng MAX toàn ảnh: lỗi nhỏ, nằm ở đâu cũng được "nhìn thấy"."""

    def __init__(self):
        super().__init__()
        def khoi(vao, ra):
            return [nn.Conv2d(vao, ra, 3, padding=1), nn.BatchNorm2d(ra), nn.ReLU()]
        self.dac_trung = nn.Sequential(*khoi(1, 8), nn.MaxPool2d(2), *khoi(8, 16), nn.MaxPool2d(2),
                                       *khoi(16, 32), nn.AdaptiveMaxPool2d(1))
        self.dau_ra = nn.Linear(32, 1)

    def forward(self, x):
        return self.dau_ra(self.dac_trung(x).flatten(1)).squeeze(1)       # logit; sigmoid → điểm NG


def tien_xu_ly(X):
    """uint8 (N, 48, 48) → float32 (N, 1, 48, 48): trừ trung bình từng ảnh, chia hằng số 32.
    Hằng số (không chia độ lệch chuẩn) để giữ thông tin tương phản; ghi vào metadata khi triển khai."""
    x = X.astype(np.float32)[:, None]
    return (x - x.mean(axis=(2, 3), keepdims=True)) / 32.0


def diem_ng(mo_hinh, X):
    mo_hinh.eval()
    with torch.no_grad():
        return torch.sigmoid(mo_hinh(torch.from_numpy(tien_xu_ly(X)))).numpy()


mo_hinh = MangNho()
SO_EPOCH = 25
opt = torch.optim.Adam(mo_hinh.parameters(), lr=3e-3)

TRIGGER, READY, BUSY, DONE = 0, 1, 2, 3                 # coil
HR_MA_SP, HR_CHU_KY, HR_KQ = 0, 1, 10                   # HR 10 KQ, 11 chu kỳ, 12 mã lỗi, 13 phiên bản, 14–15 điểm
KQ_OK, KQ_NG, KQ_LOI = 1, 2, 3


class PlcGia:
    """PLC giả (pymodbus 3.15, SimData/SimDevice) như bài Modbus TCP và Capstone 1."""

    def __init__(self, port):
        self.port = port

    def bat(self):
        self._da_nghe = threading.Event()
        self._thread = threading.Thread(target=lambda: asyncio.run(self._main()), daemon=True)
        self._thread.start()
        if not self._da_nghe.wait(timeout=3):
            raise RuntimeError("PLC giả không khởi động được")

    async def _main(self):
        self.loop = asyncio.get_running_loop()
        bo_nho = SimDevice(id=1, simdata=(
            [SimData(0, count=16, values=False, datatype=DataType.BITS)],
            [SimData(0, count=16, values=False, datatype=DataType.BITS)],
            [SimData(0, count=32, values=0, datatype=DataType.REGISTERS)],
            [SimData(0, count=16, values=0, datatype=DataType.REGISTERS)],
        ))
        self.server = ModbusTcpServer(bo_nho, address=("127.0.0.1", self.port))
        await self.server.serve_forever(background=True)
        self._da_nghe.set()
        await self.server.serving

    def tat(self):
        asyncio.run_coroutine_threadsafe(self.server.shutdown(), self.loop).result(timeout=3)
        self._thread.join(timeout=3)

    def _goi(self, coro):
        return asyncio.run_coroutine_threadsafe(coro, self.loop).result(timeout=2)

    def doc_coil(self, dc):
        return self._goi(self.server.async_getValues(1, 1, dc, 1))[0]

    def ghi_coil(self, dc, v):
        self._goi(self.server.async_setValues(1, 5, dc, [bool(v)]))

    def doc_hr(self, dc, n=1):
        return self._goi(self.server.async_getValues(1, 3, dc, n))

    def ghi_hr(self, dc, gia_tri):
        self._goi(self.server.async_setValues(1, 16, dc, list(gia_tri)))


def cho(dk, timeout):
    han = time.monotonic() + timeout
    while time.monotonic() < han:
        if dk():
            return True
        time.sleep(0.002)
    return False


def chuong_trinh_plc(plc, so_sp, nhan):
    for k in range(1, so_sp + 1):
        if not cho(lambda: plc.doc_coil(READY), 3.0):
            nhan.append((k, "KHONG_READY", None))
            continue
        plc.ghi_hr(HR_MA_SP, [301, k])
        plc.ghi_coil(TRIGGER, 1)
        if cho(lambda: plc.doc_coil(DONE), 1.0):
            r = plc.doc_hr(HR_KQ, 6)
            diem = ModbusTcpClient.convert_from_registers(r[4:6], ModbusTcpClient.DATATYPE.FLOAT32)
            nhan.append((k, r[0] if r[1] == k else "LECH", round(diem, 3)))
        else:
            nhan.append((k, "TIMEOUT", None))
        plc.ghi_coil(TRIGGER, 0)
        cho(lambda: not plc.doc_coil(DONE), 1.0)


class BoSuyLuan:
    """Chỉ phụ thuộc file .onnx + .json: KHÔNG import torch trên máy trạm."""

    def __init__(self, file_onnx, file_json):
        with open(file_json, encoding="utf-8") as f:
            self.meta = json.load(f)
        tc = ort.SessionOptions()
        tc.intra_op_num_threads = 1
        self.phien = ort.InferenceSession(file_onnx, tc, providers=["CPUExecutionProvider"])
        self.phien.run(None, {"anh": np.zeros((1, 1, S, S), np.float32)})   # chạy mồi: lần đầu luôn chậm

    def diem(self, cac_o):
        x = cac_o.astype(np.float32)[:, None]
        x = (x - x.mean(axis=(2, 3), keepdims=True)) / 32.0              # ĐÚNG như lúc huấn luyện
        return 1 / (1 + np.exp(-self.phien.run(None, {"anh": x})[0]))

    return tg

  • Độ sáng trung bình ảnh (EWMA, giới hạn 4σ học từ 200 sản phẩm đầu).
  • Độ nét: log của phương sai Laplacian, cũng theo dõi bằng EWMA. Chỉ tính trên ảnh được chấm OK, vì vết lỗi làm tăng độ "nét" giả (thử nghiệm ban đầu tính trên mọi ảnh đã gây một cảnh báo nhầm ở giai đoạn bình thường).

# ===== giam_sat.py — giám sát trôi dữ liệu đầu vào (drift) và tỉ lệ NG trong sản xuất =====
from collections import deque

import cv2
import numpy as np


class EWMA:
    """Biểu đồ EWMA: học trung bình và độ lệch chuẩn ở giai đoạn tham chiếu, rồi báo khi trôi quá k·σ_EWMA."""

    def __init__(self, so_tham_chieu=200, lam=0.1, k=4.0):
        self.ref, self.n, self.lam, self.k, self.z = [], so_tham_chieu, lam, k, None

    def cap_nhat(self, x):
        if len(self.ref) < self.n:
            self.ref.append(x)
            if len(self.ref) == self.n:
                self.mu, sd = np.mean(self.ref), np.std(self.ref, ddof=1)
                self.gioi_han = self.k * sd * np.sqrt(self.lam / (2 - self.lam))
                self.z = self.mu
            return None
        self.z = self.lam * x + (1 - self.lam) * self.z
        return abs(self.z - self.mu) > self.gioi_han


class GiamSat:
    def __init__(self, p0=0.03, cua_so=300):
        self.sang, self.net = EWMA(), EWMA()
        self.kq, self.p0 = deque(maxlen=cua_so), p0      # p0: tỉ lệ NG bình thường của dây chuyền
        self.lan_dau = {}

    def cap_nhat(self, i, anh, ng):
        self.kq.append(ng)
        if ng:
            return                         # thống kê đầu vào chỉ lấy trên ảnh OK: vết lỗi làm tăng độ "nét" giả
        if self.sang.cap_nhat(float(anh.mean())):
            self.lan_dau.setdefault("độ sáng", i)
        do_net = float(cv2.Laplacian(anh, cv2.CV_32F).var())   # phương sai Laplacian: thước đo độ nét
        if self.net.cap_nhat(np.log(do_net)):
            self.lan_dau.setdefault("độ nét", i)
        if len(self.kq) == self.kq.maxlen:                     # tỉ lệ NG, kiểm HAI phía
            p, sai = np.mean(self.kq), 3 * np.sqrt(self.p0 * (1 - self.p0) / len(self.kq))
            if p > self.p0 + sai or p < self.p0 - sai:
                self.lan_dau.setdefault("tỉ lệ NG", i)

Bài 1. Mỗi nắp có 64 ô. Overkill theo ô là 0,1%, các ô độc lập. Overkill theo nắp là bao nhiêu? Muốn overkill theo nắp ≤ 1% thì overkill theo ô phải nhỏ hơn bao nhiêu?

Xem đáp án

P(nắp OK bị loại) = 1 − (1 − 0,001)64 ≈ 6,2%. Muốn ≤ 1%: 1 − (1 − p)64 ≤ 0,01 → p ≤ 1 − 0,991/64 ≈ 0,016% theo ô. Đây là lý do tiêu chí phải viết theo sản phẩm, và đánh giá phải tính lại từ ô sang sản phẩm.

Bài 2. Khách hàng yêu cầu chứng minh escape dưới 1% với độ tin cậy khoảng 95%. Theo quy tắc 3/n, cần bao nhiêu mẫu NG (độc lập) chạy qua trạm mà không lọt cái nào?

Xem đáp án

3/n ≤ 0,01 → n ≥ 300 mẫu NG không lọt mẫu nào. Tập test của bài chỉ có 89 ô NG, nên chỉ chứng minh được dưới khoảng 3,4%. Cần thu thập thêm mẫu NG thật (hoặc nhiều lần chạy của các mẫu khác nhau, cẩn thận với giả định độc lập).

Bài 3. Sau khi nhà cung cấp nhôm đổi, ảnh lô mới sáng hơn 15% và vân thô hơn. Bộ giám sát độ sáng báo ngay. Bạn làm gì, theo thứ tự?

(1) Không tắt cảnh báo; chạy mẫu chuẩn OK/NG của ca để biết trạm còn phân loại đúng không. (2) Lấy mẫu vài trăm sản phẩm lô mới, QC gán nhãn, đánh giá escape/overkill của mô hình hiện tại trên đó. (3) Nếu không đạt: thêm dữ liệu lô mới vào train (giữ một phần làm test), huấn luyện lại, chọn ngưỡng, sinh biên bản, tăng phiên bản. (4) Cập nhật giai đoạn tham chiếu của bộ giám sát cho vật liệu mới. Trong lúc chờ, tăng tần suất kiểm tra thủ công lô đó.

Tóm tắt

  • Chia dữ liệu theo lô; chọn ngưỡng trên val bằng quy tắc viết trước (escape trước, overkill trong ngân sách); test chỉ dùng một lần.
  • Ngưỡng 0,5 và "độ chính xác" không phải tiêu chí nhà máy: hãy báo escape, overkill và giới hạn thống kê của chúng.
  • Triển khai bằng ONNX + metadata (phiên bản, ngưỡng, tiền xử lý), kiểm tra sai số và khoảng cách tới ngưỡng, chạy mồi, đo p50/p99 từng bước.
  • Tích hợp như mọi trạm: ghi kết quả trước DONE, không có ảnh thì LỖI, lưu ảnh NG có điểm và phiên bản.
  • Mô hình hỏng im lặng: giám sát đầu vào (độ sáng, độ nét) phát hiện drift sớm hơn tỉ lệ NG; kiểm hai phía và dùng mẫu chuẩn mỗi ca.
  • Ba capstone cùng một khung: đặc tả bằng con số, lõi kỹ thuật có đáp án, tích hợp PLC, biên bản tự sinh.
