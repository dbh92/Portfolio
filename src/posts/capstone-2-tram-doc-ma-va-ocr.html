Capstone 1 đo micromet; dự án này đo bằng một thước khác: tỉ lệ đọc đúng, số hộp lọt lỗi và tốc độ dây chuyền. Bạn xây trạm kiểm tra nhãn trên dây chuyền đóng hộp: trong một chu kỳ, đọc mã 2D, đọc dòng số seri in bên cạnh (OCR), xác minh chất lượng in (OCV), đối chiếu với lệnh sản xuất (định dạng, chữ số kiểm tra, dải seri, chống trùng bằng CSDL) và trả OK/NG kèm lý do trong ngân sách thời gian của dây chuyền. 1. Bối cảnh

Một nhà máy lắp ráp bộ nguồn 12 V 60 W đóng mỗi bộ nguồn vào một hộp giấy. Máy in nhãn nhiệt dán lên mỗi hộp một nhãn gồm: mã 2D chứa mã sản phẩm (PN), số seri (SN) và số lô (LOT); dòng chữ số seri in to bên cạnh cho người đọc; dòng PN và LOT cỡ nhỏ. Khách hàng (một hãng phân phối) yêu cầu: không bao giờ có hai hộp cùng số seri, chữ in phải khớp mã, và mọi số seri phải truy được về lệnh sản xuất. Quý trước có hai sự cố: máy in bị lỗi phần mềm in lại một dải seri cũ sau khi khởi động lại, và đầu in bị tắc một dải làm chữ mất nét mà không ai để ý suốt nửa ca.

6.2. Mã khởi đầu: nhãn có đáp án và lỗi biết trước

Bộ sinh nhãn vẽ nhãn thẳng 500 × 200 px (module mã 5 px, chữ seri bước cố định 22 px như máy in nhãn công nghiệp), rồi "dán" lên hộp trong ảnh 640 × 480 với độ lệch, góc xoay, độ mờ, tương phản, nhiễu ngẫu nhiên. Tham số loi tiêm lỗi in: mã bị xước, giọt mực, đầu in tắc, nhãn mờ. Hàm chu_so_luhn tính chữ số kiểm tra, thử bằng ví dụ chuẩn của thuật toán Luhn.

# ===== sinh_nhan.py — "camera giả": ảnh nhãn hộp có đáp án và lỗi biết trước =====
import math

import cv2
import numpy as np

W_ANH, H_ANH = 640, 480
W_NHAN, H_NHAN = 500, 200
FONT, CO_CHU, NET = cv2.FONT_HERSHEY_SIMPLEX, 0.9, 2   # máy in nhãn: font cố định, bước chữ cố định
BUOC_CHU, X_SERI, Y_SERI = 22, 200, 112                # ô chữ rộng 22 px, chân dòng seri tại y = 112
QR = cv2.QRCodeEncoder.create()                        # ĐỨNG THAY DataMatrix trong bản demo (xem mục 6)


def chu_so_luhn(chu_so):
    """Chữ số kiểm tra Luhn cho một chuỗi chữ số."""
    tong = 0
    for i, c in enumerate(reversed(chu_so)):
        d = int(c) * (2 if i % 2 == 0 else 1)            # nhân đôi từ chữ số sát chữ số kiểm tra
        tong += d - 9 if d > 9 else d
    return str((10 - tong % 10) % 10)


def tao_seri(yymm, line, stt):
    """VN + YYMM + mã line + 5 chữ số thứ tự + 1 chữ số kiểm tra Luhn → 13 ký tự."""
    than = f"{yymm}{stt:05d}"
    return f"VN{yymm}{line}{stt:05d}{chu_so_luhn(than)}"


def ve_nhan(pn, seri, lot, seri_in=None):
    """Nhãn thẳng (hệ tọa độ nhãn). seri_in: chữ in khác mã (mô phỏng lỗi dữ liệu máy in)."""
    nhan = np.full((H_NHAN, W_NHAN), 235, np.uint8)
    ma = QR.encode(f"PN:{pn};SN:{seri};LOT:{lot}")    # có sẵn vùng trắng 2 module quanh mã
    ma = cv2.resize(ma, None, fx=5, fy=5, interpolation=cv2.INTER_NEAREST)   # module 5 px
    nhan[18:18 + ma.shape[0], 20:20 + ma.shape[1]] = np.where(ma < 128, 25, 235)
    cv2.putText(nhan, f"PN {pn}", (X_SERI, 50), FONT, 0.6, 25, 1, cv2.LINE_AA)
    for i, c in enumerate(seri_in or seri):              # bước chữ cố định: ô thứ i bắt đầu ở X_SERI + i·22
        (w, _), _ = cv2.getTextSize(c, FONT, CO_CHU, NET)
        cv2.putText(nhan, c, (X_SERI + i * BUOC_CHU + (BUOC_CHU - w) // 2, Y_SERI), FONT, CO_CHU, 25, NET,
                    cv2.LINE_AA)
    cv2.putText(nhan, f"LOT {lot}", (X_SERI, 150), FONT, 0.6, 25, 1, cv2.LINE_AA)
    return nhan


def chup(nhan, rng, loi=None):
    """Đặt nhãn lên hộp trong ảnh camera: lệch, xoay, mờ, nhiễu; tiêm lỗi in nếu có."""
    nhan = nhan.copy()
    if loi == "ma_hong":                                 # vết xước/bẩn che góc mã
        cv2.circle(nhan, (60, 58), 30, 235, -1)
    elif loi == "cham_muc":                              # giọt mực trên một ký tự của số seri
        x = X_SERI + int(rng.integers(0, 13)) * BUOC_CHU + 11
        cv2.circle(nhan, (x, Y_SERI - 10), 7, 25, -1)
    elif loi == "thieu_net":                             # đầu in tắc: mất một dải ngang qua dòng seri
        nhan[Y_SERI - 12:Y_SERI - 7, X_SERI:X_SERI + 13 * BUOC_CHU] = 235
    goc = rng.uniform(-4, 4)
    tam = (W_ANH / 2 + rng.uniform(-15, 15), H_ANH / 2 + rng.uniform(-15, 15))
    M = cv2.getRotationMatrix2D((W_NHAN / 2, H_NHAN / 2), goc, 1.0)
    M[:, 2] += np.array(tam) - (W_NHAN / 2, H_NHAN / 2)
    anh = cv2.warpAffine(nhan, M, (W_ANH, H_ANH), flags=cv2.INTER_LINEAR, borderValue=90)  # nền hộp
    mo, tuong_phan = (1.3, 0.55) if loi == "mo" else (rng.uniform(0.5, 0.9), rng.uniform(0.85, 1.0))
    anh = cv2.GaussianBlur(anh.astype(np.float32), (0, 0), mo)
    anh = 128 + (anh - 128) * tuong_phan + rng.normal(0, 4, anh.shape)
    return np.clip(anh, 0, 255).astype(np.uint8)

  3. Căn dòng: sai số vài pixel ở góc mã bị khuếch đại ở cuối dòng chữ; so cả dòng mong đợi trong vùng ±8 px để tìm độ dời trước khi đọc từng ô.
# ===== doc_nhan.py — đọc mã 2D, nắn nhãn về hệ chuẩn, OCR + OCV dòng số seri =====
import string
import cv2
import numpy as np

# Bản demo dùng QR của OpenCV thay cho DataMatrix. Hai bộ dò của OpenCV mạnh yếu khác nhau,
# nên đọc theo "bậc thang": thử cách nhanh trước, chỉ thử tiếp khi thất bại (như đầu đọc thương mại).
BO_DO_MA = (cv2.QRCodeDetector(), cv2.QRCodeDetectorAruco())


def doc_ma(anh):
    """Trả (nội dung, 4 góc, bậc đã dùng) hoặc (None, None, None)."""
    _, otsu = cv2.threshold(anh, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    for bac, (bo_do, x) in enumerate(((BO_DO_MA[0], anh), (BO_DO_MA[1], anh),
                                      (BO_DO_MA[0], otsu), (BO_DO_MA[1], otsu))):
        noi_dung, goc, _ = bo_do.detectAndDecode(x)
        if noi_dung and goc is not None:
            return noi_dung, goc.reshape(4, 2).astype(np.float32), bac
    return None, None, None


def tach_truong(noi_dung):
    """'PN:...;SN:...;LOT:...' → dict. Thiếu trường bắt buộc → None."""
    try:
        d = dict(p.split(":", 1) for p in noi_dung.split(";"))
    except ValueError:
        return None
    return d if {"PN", "SN", "LOT"} <= d.keys() else None


def nan_nhan(anh, goc_ma):
    """Dùng chính mã 2D làm mốc định vị: phép affine từ 3 góc ảnh → nhãn chuẩn 500 × 200.
    Chỉ dùng góc trên-trái, trên-phải, dưới-trái (có mẫu định vị, rất ổn định); góc dưới-phải của QR
    là góc ước lượng, sai lệch vài pixel. DataMatrix cũng vậy: hai cạnh liền chữ L là mốc tin cậy."""
    chon = [0, 1, 3]
    A = cv2.getAffineTransform(goc_ma[chon], GOC_MA_CHUAN[chon].astype(np.float32))
    return cv2.warpAffine(anh, A, (W_NHAN, H_NHAN), flags=cv2.INTER_LINEAR, borderValue=235)


def can_dong(nhan, mong_doi, bien=8):
    """Tinh chỉnh vị trí dòng seri: so cả dòng mong đợi (ghép từ mẫu ký tự) trong vùng ±bien px.
    Sai số góc mã vài pixel bị khuếch đại ở cuối dòng chữ (xa mã gấp 3 lần cạnh mã)."""
    o = CFG["o_seri"]
    dong = np.hstack([MAU[c] for c in mong_doi])
    vung = nhan[o["y_tren"] - bien:o["y_tren"] + o["cao"] + bien, o["x"] - bien:o["x"] + dong.shape[1] + bien]
    _, diem, _, (dx, dy) = cv2.minMaxLoc(cv2.matchTemplate(vung, dong, cv2.TM_CCOEFF_NORMED))
    return dx - bien, dy - bien


def doc_seri(nhan, mong_doi):
    """OCR theo ô cố định + OCV so với chuỗi mong đợi (lấy từ mã 2D).
    Trả (chuỗi đọc được, NCC từng ký tự, diện tích lỗi in từng ký tự, vị trí kém tin cậy)."""
    o, le = CFG["o_seri"], CFG["le_tim_px"]
    dx, dy = can_dong(nhan, mong_doi)
    doc, ncc, loi, kem = [], [], [], []
    for i, loai in enumerate(CFG["dinh_dang_seri"]):
        x0, y0 = o["x"] + i * o["buoc"] + dx, o["y_tren"] + dy
        o_chu = nhan[y0 - le:y0 + o["cao"] + le, x0 - le:x0 + o["buoc"] + le]
        # OCR: ký tự khớp nhất, chỉ trong tập ký tự mà định dạng cho phép ở vị trí này
        ung_vien = string.ascii_uppercase if loai == "L" else string.digits
        diem = {c: float(cv2.matchTemplate(o_chu, MAU[c], cv2.TM_CCOEFF_NORMED).max()) for c in ung_vien}
        xep = sorted(diem, key=diem.get, reverse=True)
        doc.append(xep[0])
        if diem[xep[0]] - diem[xep[1]] < CFG["bien_ocr"]:
            kem.append(i)                                    # hai ứng viên quá sát nhau: không chắc
        # OCV: so với ký tự MONG ĐỢI. (1) NCC; (2) diện tích pixel lệch sau khi chỉnh sáng/tương phản
        mau = MAU[mong_doi[i]]
        r = cv2.matchTemplate(o_chu, mau, cv2.TM_CCOEFF_NORMED)
        _, s, _, (px, py) = cv2.minMaxLoc(r)
        p = o_chu[py:py + o["cao"], px:px + o["buoc"]].astype(np.float32)
        a, b = np.polyfit(p.ravel(), mau.ravel().astype(np.float32), 1)   # đưa về cùng thang xám với mẫu
        ncc.append(s)
        loi.append(int((np.abs(a * p + b - mau) > CFG["lech_muc_xam"]).sum()))
    return "".join(doc), ncc, loi, kem

# ===== tram_ma.py — một chu kỳ: đọc mã + OCR/OCV + xác minh với lệnh sản xuất + chống trùng =====
import re
import sqlite3
import time
from datetime import datetime
from pathlib import Path

import numpy as np

LENH_SX = {"po": "PO-2610-0815", "pn": "PSU-12V-60W", "lot": "L2610A", "yymm": "2610", "line": "A",
           "stt_dau": 100, "stt_cuoi": 599}                   # 500 hộp, nhận từ MES khi đổi lệnh
MAU_SERI = re.compile(r"^VN(\d{4})([A-Z])(\d{5})(\d)$")

SCHEMA = """
CREATE TABLE IF NOT EXISTS seri_da_dung (seri TEXT PRIMARY KEY, po TEXT NOT NULL, thoi_gian TEXT NOT NULL);
CREATE TABLE IF NOT EXISTS ket_qua (id INTEGER PRIMARY KEY, thoi_gian TEXT, po TEXT, seri_ma TEXT,
    seri_chu TEXT, ket_qua TEXT CHECK (ket_qua IN ('OK', 'NG')), ly_do TEXT, ncc_min REAL, loi_in_max INTEGER,
    bac_doc INTEGER,
    ms REAL);
"""


class TramMa:
    def __init__(self, lenh, csdl):
        self.lenh = lenh
        self.con = sqlite3.connect(csdl)
        self.con.execute("PRAGMA journal_mode=WAL")
        self.con.executescript(SCHEMA)
        self.stt_cao_nhat = None
        self.nhay_coc = []                                   # các lần số thứ tự nhảy: cảnh báo, không NG

    def _xac_minh_ma(self, noi_dung):
        """Kiểm tra nội dung mã với lệnh sản xuất. Trả (lý do NG hoặc None, seri, stt)."""
        tr = tach_truong(noi_dung)
        if tr is None:
            return "noi_dung_ma_sai", None, None
        sn, m = tr["SN"], MAU_SERI.match(tr["SN"])
        if not m:
            return "sai_dinh_dang", sn, None
        yymm, line, stt, kt = m.groups()
        if tr["PN"] != self.lenh["pn"] or tr["LOT"] != self.lenh["lot"] or (yymm, line) != (
                self.lenh["yymm"], self.lenh["line"]):
            return "sai_lenh_sx", sn, None
        if chu_so_luhn(yymm + stt) != kt:
            return "sai_checksum", sn, None
        stt = int(stt)
        if not self.lenh["stt_dau"] <= stt <= self.lenh["stt_cuoi"]:
            return "ngoai_dai_seri", sn, None
        return None, sn, stt

    def xu_ly(self, anh):
        t = [time.perf_counter()]
        noi_dung, goc, bac = doc_ma(anh)
        t.append(time.perf_counter())
        sn = chu = ncc_min = loi_max = None
        if noi_dung is None:
            ly_do = "khong_doc_duoc_ma"
        else:
            ly_do, sn, stt = self._xac_minh_ma(noi_dung)
            if stt is not None:                              # theo dõi thứ tự in (chỉ để truy vết)
                if self.stt_cao_nhat is not None and stt > self.stt_cao_nhat + 1:
                    self.nhay_coc.append((self.stt_cao_nhat, stt))
                self.stt_cao_nhat = max(stt, self.stt_cao_nhat or stt)
            if sn is not None and MAU_SERI.match(sn):        # OCR + OCV dòng chữ, kể cả khi đã NG
                chu, ncc, dt_loi, kem = doc_seri(nan_nhan(anh, goc), mong_doi=sn)
                ncc_min, loi_max = round(min(ncc), 3), max(dt_loi)
                if ly_do is None and chu != sn and not kem:
                    ly_do = "chu_khac_ma"                    # chữ in rõ ràng nhưng KHÁC nội dung mã
                elif ly_do is None and (chu != sn or ncc_min < CFG["nguong_ocv"]
                                        or loi_max > CFG["dien_tich_loi_max"]):
                    ly_do = "in_kem"                         # chữ in không đạt chất lượng
        t.append(time.perf_counter())
        thoi_gian = datetime.now().isoformat(timespec="milliseconds")
        if ly_do is None:
            try:                                             # chống trùng: khóa chính của SQLite
                self.con.execute("INSERT INTO seri_da_dung VALUES (?, ?, ?)", (sn, self.lenh["po"], thoi_gian))
            except sqlite3.IntegrityError:
                ly_do = "trung_seri"
        ms = (time.perf_counter() - t[0]) * 1000
        self.con.execute("INSERT INTO ket_qua (thoi_gian, po, seri_ma, seri_chu, ket_qua, ly_do, ncc_min, loi_in_max,"
                         " bac_doc, ms) VALUES (?,?,?,?,?,?,?,?,?,?)",
                         (thoi_gian, self.lenh["po"], sn, chu, "OK" if ly_do is None else "NG", ly_do, ncc_min,
                          loi_max, bac, round(ms, 2)))
        self.con.commit()
        t.append(time.perf_counter())
        return {"ket_qua": "OK" if ly_do is None else "NG", "ly_do": ly_do, "seri": sn, "chu": chu,
                "ncc_min": ncc_min, "loi_max": loi_max, "bac": bac,
                "ms": {"doc_ma": (t[1] - t[0]) * 1e3, "ocr_ocv": (t[2] - t[1]) * 1e3, "csdl": (t[3] - t[2]) * 1e3}}

6.5. Trên máy của bạn: DataMatrix và Tesseract/PaddleOCR

Hai khối dưới đây là phần thay thế cho trạm thật, viết theo cùng cách dùng như bài đọc mã trên bao bì và bài OCR với Tesseract. Chúng không chạy trong môi trường thử của bài (thiếu libdmtx và Tesseract), hãy chạy trên máy bạn sau khi cài.

# (minh họa) — chạy trên máy bạn sau khi cài: pip install pylibdmtx (+ libdmtx của hệ điều hành)
import cv2
import numpy as np
from pylibdmtx.pylibdmtx import decode as dm_decode
from pylibdmtx.pylibdmtx import encode as dm_encode

# 1) Sinh nhãn thử có DataMatrix: thay hai dòng QR.encode/resize trong sinh_nhan.py
enc = dm_encode("PN:PSU-12V-60W;SN:VN2610A001238;LOT:L2610A".encode("utf-8"))
dm = np.frombuffer(enc.pixels, np.uint8).reshape(enc.height, enc.width, 3)[:, :, 0]   # ảnh RGB → xám
print("Kích thước ảnh mã (px):", dm.shape)


# 2) Thay doc_ma(): đọc DataMatrix trong vùng quan tâm, luôn đặt timeout
def doc_ma_dm(anh_xam):
    kq = dm_decode(anh_xam, timeout=100, max_count=1)       # timeout tính bằng mili giây
    if not kq:
        return None, None
    r = kq[0]
    x, y, w, h = r.rect.left, r.rect.top, r.rect.width, r.rect.height
    y = anh_xam.shape[0] - y - h                            # libdmtx đặt gốc tọa độ ở dưới-trái
    return r.data.decode("utf-8", "replace"), (x, y, w, h)

pylibdmtx trả về nội dung và một hình chữ nhật rect (vị trí, kích thước) của mã, không trả đủ 3 góc mốc như bộ đọc QR của OpenCV. Để nắn nhãn khi nhãn xoay, bạn có hai cách: tìm hướng của mã bằng minAreaRect trên vùng mã quanh rect rồi xác định cạnh chữ L liền (cạnh có mức đen liên tục), hoặc định vị nhãn bằng template matching như dự án đọc số seri trên linh kiện. Các đầu đọc và thư viện thương mại thường trả đủ góc và hướng của mã. Luôn đặt timeout và đọc trong vùng quan tâm nhỏ: libdmtx có thể rất chậm trên ảnh lớn không có mã.

# (minh họa) — cần cài chương trình Tesseract + pip install pytesseract (xem bài OCR với Tesseract)
import cv2
import pytesseract

WL = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"


def doc_seri_tesseract(nhan):
    """Thay phần OCR trong doc_seri(): cắt cả dòng seri trên nhãn đã nắn, phóng to, nhị phân, đọc 1 dòng."""
    o = CFG["o_seri"]
    roi = nhan[o["y_tren"] - 6:o["y_tren"] + o["cao"] + 6, o["x"] - 8:o["x"] + o["so_ky_tu"] * o["buoc"] + 8]
    roi = cv2.resize(roi, None, fx=2, fy=2, interpolation=cv2.INTER_CUBIC)
    _, th = cv2.threshold(roi, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    chu = pytesseract.image_to_string(th, config=f"--psm 7 -c tessedit_char_whitelist={WL}")
    return chu.strip().replace(" ", "")

Kết quả Tesseract/PaddleOCR đi tiếp qua bước sửa nhầm lẫn theo định dạng (bảng TO_DIGIT/TO_ALPHA của dự án đọc số seri), rồi vào đúng các bước xác minh của TramMa. OCV (NCC và diện tích lỗi in theo mẫu ký tự cắt từ nhãn thật) vẫn giữ, vì OCR không cho biết chữ in đẹp hay xấu.

7.1. Bộ nhãn vàng 300 hộp

Bộ đánh giá mô phỏng 300 hộp liên tiếp của lệnh sản xuất, với 8 loại lỗi rải ở các vị trí cố định (trạm không biết), 18 nhãn mờ tương phản thấp trong nhóm nhãn tốt để thử overkill, và 4 lần máy in bỏ qua một số thứ tự (không phải lỗi, chỉ cảnh báo).   • Thời gian tùy máy và tùy tải. Các nhãn cần đọc lại ở bậc cao tốn thêm thời gian, nên ngân sách phải tính theo p99, không theo trung bình. Nếu p99 trên PC của trạm vượt ngân sách, cách đầu tiên là thu nhỏ vùng đọc mã quanh vị trí nhãn dự kiến, sau đó mới tính tới xử lý song song nhiều hộp.
300 nhãn không chứng minh được "0 ppm"0 lỗi trên 256 nhãn tốt chỉ cho biết tỉ lệ đọc lỗi có lẽ dưới khoảng 1,2% (quy tắc 3/n đã học ở bài tiêu chí OK/NG). Trên dây chuyền thật, hãy ghi tỉ lệ đọc và tỉ lệ dùng từng bậc đọc theo từng ca, và nghiệm thu SAT bằng vài nghìn hộp thật.

Bài 1. Tính chữ số kiểm tra Luhn cho phần số 261000123 bằng tay, rồi kiểm lại bằng chu_so_luhn. Luhn có phát hiện được lỗi đổi chỗ hai chữ số kề nhau "00123" → "00213" không?

Xem đáp án

Từ phải sang trái, nhân đôi các chữ số ở vị trí lẻ (sát chữ số kiểm tra): 3→6, 2, 1→2, 0, 0→0, 0, 1→2, 6, 2→4. Tổng = 6 + 2 + 2 + 0 + 0 + 0 + 2 + 6 + 4 = 22, chữ số kiểm tra = (10 − 2) mod 10 = 8, đúng như seri mẫu VN2610A001238. Với "00213": các chữ số 3→6, 1, 2→4, 0, 0, 0, 1→2, 6, 2→4, tổng 23, chữ số kiểm tra 7 ≠ 8, nên phát hiện được. Luhn phát hiện mọi lỗi sai một chữ số và hầu hết lỗi đổi chỗ hai chữ số kề nhau (trừ cặp 09 ↔ 90).

Bài 2. Dây chuyền tăng lên 200 hộp/phút. Ngân sách 150 ms còn hợp lý không? Nếu p99 đo được là 110 ms nhưng có 0,3% chu kỳ vượt 300 ms do ghi đĩa, bạn làm gì?

Xem đáp án

200 hộp/phút = 300 ms/hộp. 150 ms vẫn để lại một nửa cho truyền ảnh, giao tiếp và dao động, nên vẫn hợp lý. Nhưng 0,3% chu kỳ vượt 300 ms nghĩa là cứ khoảng 330 hộp lại có một hộp không có kết quả kịp: không chấp nhận được. Chuyển ghi CSDL và ảnh sang thread nền với hàng đợi giới hạn (như dự án trạm vision mô phỏng), đo lại; đồng thời PLC phải coi "không có kết quả đúng hạn" là NG.

Bài 3. Vì sao trạm đọc mã trước rồi dùng nội dung mã làm "đáp án" cho OCV, mà không làm ngược lại?

Xem đáp án

Mã 2D có sửa lỗi Reed-Solomon và kiểm tra tính toàn vẹn khi giải mã: đọc được thì gần như chắc chắn đúng. OCR không có cơ chế đó, có thể đọc sai mà vẫn "tự tin". Dùng chuỗi đáng tin hơn làm chuẩn để kiểm chuỗi kém tin hơn. Nếu mã không đọc được, trạm không có chuẩn, nên NG chứ không "tin tạm" OCR.

Tóm tắt

  • Một chu kỳ: đọc mã (bậc thang thử lại) → xác minh nội dung với lệnh SX → nắn nhãn theo mã → OCR + OCV → chống trùng bằng khóa chính CSDL → ghi truy vết.
  • Mã 2D vừa là mốc định vị vừa là đáp án cho OCV; chỉ dùng các góc mốc tin cậy, căn dòng trước khi đọc ô.
  • OCR đọc đúng chưa đủ: OCV từng ký tự (NCC, diện tích lỗi in) mới bắt được chữ mất nét, giọt mực.
  • Phân biệt NG theo lý do: chữ khác mã (lỗi dữ liệu) khác chữ in kém (lỗi đầu in); chống trùng là bước cuối.
  • Tốc độ đánh giá bằng p99 trên đúng máy của trạm; nhãn cần đọc lại là nhãn chậm.
  • Bản demo dùng QR và OCR theo mẫu thay cho DataMatrix và Tesseract; trên máy bạn, thay đúng hai hàm đó và giữ nguyên bộ đánh giá.
