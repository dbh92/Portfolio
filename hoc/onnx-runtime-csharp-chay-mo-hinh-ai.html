Ở bài trước "Xuất ONNX và kiểm tra kết quả khớp PyTorch", bạn đã có một file .onnx được kiểm chứng trong Python. Nhưng phần mềm trạm kiểm tra ở nhiều nhà máy Việt Nam lại viết bằng C# (WinForms, WPF, hoặc dịch vụ .NET chạy cạnh SDK camera và thư viện PLC). Cầu nối là ONNX Runtime cho .NET: cùng lõi C++ với bản Python, gọi trực tiếp từ C#, không cần cài Python trên máy công nghiệp. Khó khăn thật sự không nằm ở dòng gọi mô hình, mà ở chỗ viết lại tiền xử lý cho giống hệt lúc huấn luyện và giải mã đầu ra cho đúng. Bài này đi trọn đường đó: chuẩn bị bộ ảnh vàng trong Python, viết ứng dụng console C# đọc ảnh bằng OpenCvSharp, chạy suy luận, đo thời gian, tự so với bộ ảnh vàng, rồi giải mã đầu ra phân loại lẫn phát hiện vật. Bài sau "Tăng tốc suy luận: OpenVINO, TensorRT, FP16, INT8" sẽ làm cho đúng chương trình này chạy nhanh hơn.

Mục tiêu cuối bàiCài đúng gói NuGet (CPU hay GPU); tạo InferenceSession một lần với SessionOptions phù hợp; tiền xử lý trong C# cho ra tensor trùng với Python tới sai số 10-6; chạy suy luận với DenseTensor<float> + NamedOnnxValue (và biết API OrtValue mới hơn); giải phóng tài nguyên đúng cách bằng using; giải mã đầu ra phân loại (softmax, ngưỡng) và phát hiện vật (letterbox, NMS); kiểm tra tương đương (parity test) với bộ ảnh vàng do Python tạo.

1. Bức tranh tổng thể: Python huấn luyện, C# chạy trên trạm

Điểm mấu chốt: file ONNX chỉ chứa mạng. Mọi bước trước mạng (resize, đổi kênh màu, chia 255, trừ mean chia std, đổi layout) và sau mạng (softmax, ngưỡng, NMS, đổi tọa độ) nằm trong code của bạn, viết hai lần bằng hai ngôn ngữ. Vì vậy cách làm chuyên nghiệp là: Python xuất kèm một bộ ảnh vàng (golden set) có đầu ra mong đợi, và chương trình C# phải tái tạo đúng các con số đó trước khi được lắp lên dây chuyền.

2. Phía Python: mô hình nhỏ và bộ ảnh vàng

Để bài tự chạy được, ta dựng lại tình huống của bài trước ở cỡ nhỏ: ảnh camera 120×120 pixel của tấm kim loại; ảnh OK có thể có chấm mực xanh của QC, ảnh NG có vết gỉ màu cam. Mô hình chỉ khác nhau ở màu, nên sai thứ tự kênh màu sẽ lộ ra ngay. Với mô hình thật của bạn, chỉ cần thay mô hình và giữ nguyên hàm tien_xu_ly đã dùng lúc huấn luyện.

# Bài viết chạy với torch 2.14.1 (CPU), onnx 1.23.2, onnxruntime 1.31.0, onnxscript 0.7.2.
import hashlib
import json
import shutil
from pathlib import Path

import cv2
import numpy as np
import torch
import torch.nn as nn

torch.manual_seed(0)
torch.set_num_threads(4)

KICH_THUOC = 64
MEAN = np.array([0.485, 0.456, 0.406], np.float32)            # theo thứ tự R, G, B
STD = np.array([0.229, 0.224, 0.225], np.float32)


def tao_anh(ng, rng):
    """Ảnh 120x120 BGR: tấm kim loại xám; OK có thể có chấm mực xanh, NG có vết gỉ cam."""
    anh = np.full((120, 120, 3), 128, np.float32) + rng.normal(0, 8, (120, 120, 3))
    anh += np.linspace(-15, 15, 120)[None, :, None]             # ánh sáng không đều
    tam = tuple(int(v) for v in rng.integers(25, 95, 2))
    r = int(rng.integers(6, 12))
    if ng:
        cv2.circle(anh, tam, r, (40, 110, 200), -1)              # gỉ: cam (B thấp, R cao)
    elif rng.random() < 0.6:
        cv2.circle(anh, tam, r, (200, 120, 40), -1)              # mực QC: xanh dương
    return np.clip(anh, 0, 255).astype(np.uint8)


def tien_xu_ly(anh_bgr):
    """BGR uint8 (H, W, 3) -> float32 (3, 64, 64): resize INTER_AREA, BGR->RGB, /255, mean/std, HWC->CHW."""
    a = cv2.resize(anh_bgr, (KICH_THUOC, KICH_THUOC), interpolation=cv2.INTER_AREA)
    a = cv2.cvtColor(a, cv2.COLOR_BGR2RGB).astype(np.float32) / 255.0
    a = (a - MEAN) / STD
    return np.ascontiguousarray(a.transpose(2, 0, 1))


rng = np.random.default_rng(0)
nhan = np.array([i % 2 for i in range(400)])                     # 0 = OK, 1 = NG
X = torch.from_numpy(np.stack([tien_xu_ly(tao_anh(n, rng)) for n in nhan]))
y = torch.from_numpy(nhan).long()

model = nn.Sequential(
    nn.Conv2d(3, 8, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
    nn.Conv2d(8, 16, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
    nn.Conv2d(16, 16, 3, padding=1), nn.ReLU(),
    nn.AdaptiveAvgPool2d(1), nn.Flatten(), nn.Linear(16, 2))
opt = torch.optim.Adam(model.parameters(), lr=5e-3)
for ep in range(4):
    for i in range(0, 400, 32):
        loss = nn.functional.cross_entropy(model(X[i:i + 32]), y[i:i + 32])
        opt.zero_grad()
        loss.backward()
        opt.step()
model.eval()
with torch.no_grad():
    print("độ chính xác trên tập huấn luyện:", (model(X).argmax(1) == y).float().mean().item())

torch.onnx.export(model, (X[:1],), "ok_ng_cs.onnx", input_names=["images"], output_names=["logits"],
                  dynamic_shapes={"input": {0: "batch"}},       # "input" = tên tham số forward của nn.Sequential
                  opset_version=18, external_data=False, verbose=False)
print("ok_ng_cs.onnx:", Path("ok_ng_cs.onnx").stat().st_size, "byte")

Bây giờ tạo bộ ảnh vàng: 6 ảnh PNG (PNG không nén mất dữ liệu, nên hai phía đọc ra cùng pixel), kèm file golden.json ghi đặc tả tiền xử lý, mã băm mô hình và với mỗi ảnh: tổng từng kênh của tensor, giá trị pixel góc trên trái của 3 kênh, logit, xác suất NG và kết luận. Chép nguyên thư mục golden/ sang máy có chương trình C#.

import onnxruntime as ort

sess = ort.InferenceSession("ok_ng_cs.onnx", providers=["CPUExecutionProvider"])

vang.mkdir(exist_ok=True)
shutil.copy("ok_ng_cs.onnx", vang / "ok_ng_cs.onnx")

    anh = tao_anh(ng, rng_v)
    ten = f"anh_{i:02d}_{'NG' if ng else 'OK'}.png"
    cv2.imwrite(str(vang / ten), anh)

    p_ng = float(1.0 / (1.0 + np.exp(lg[0] - lg[1])))           # softmax 2 lớp

               "tong_kenh": [round(float(v), 3) for v in t.sum(axis=(1, 2))],
               "goc_tren_trai": [round(float(t[c, 0, 0]), 5) for c in range(3)],

               "p_ng": round(p_ng, 5),
               "ket_qua": "NG" if p_ng >= 0.5 else "OK"})

4. InferenceSession: tạo một lần, đọc "hợp đồng" của mô hình

InferenceSession nạp file ONNX, tối ưu đồ thị và cấp phát bộ nhớ. Việc này tốn từ vài chục mili giây tới vài giây tùy mô hình, nên chỉ làm một lần khi ứng dụng khởi động, rồi dùng lại cho mọi ảnh. InferenceSession, SessionOptions và kết quả của Run đều giữ bộ nhớ gốc (native) ngoài tầm của bộ gom rác .NET, nên đều cài IDisposable: dùng using để giải phóng đúng lúc.

using Microsoft.ML.OnnxRuntime;
using Microsoft.ML.OnnxRuntime.Tensors;

using var options = new SessionOptions
{
    GraphOptimizationLevel = GraphOptimizationLevel.ORT_ENABLE_ALL,   // gộp phép toán, bỏ nút thừa
    IntraOpNumThreads = 4,      // số luồng cho MỘT phép toán (Conv, MatMul...)
    InterOpNumThreads = 1,      // số luồng chạy song song các nhánh đồ thị (chỉ dùng khi ORT_PARALLEL)
    ExecutionMode = ExecutionMode.ORT_SEQUENTIAL
};
using var session = new InferenceSession("golden/ok_ng_cs.onnx", options);

foreach (var (name, meta) in session.InputMetadata)
    Console.WriteLine($"input  {name}: {meta.ElementType.Name} [{string.Join(", ", meta.Dimensions)}]");
foreach (var (name, meta) in session.OutputMetadata)
    Console.WriteLine($"output {name}: {meta.ElementType.Name} [{string.Join(", ", meta.Dimensions)}]");

Với file của mục 2, chương trình in đầu vào images kiểu Single (tên .NET của float) shape [-1, 3, 64, 64] và đầu ra logits shape [-1, 2]: trục động (batch) hiện là -1 trong Dimensions; tên trục (batch) nằm trong SymbolicDimensions. Hãy in metadata này mỗi khi nhận mô hình mới, và so với thẻ mô hình trước khi viết code giải mã.

Một session, nhiều luồngGọi session.Run đồng thời từ nhiều luồng là an toàn (ONNX Runtime thiết kế như vậy), nên một trạm hai camera có thể dùng chung một session. Ngược lại, đừng tạo session mới cho mỗi ảnh: vừa chậm, vừa dễ rò bộ nhớ gốc nếu quên Dispose.

5. Tiền xử lý trong C#: giống hệt Python, từng bước

Hàm tien_xu_ly của Python có 5 bước. Bảng dưới là bản dịch từng bước sang OpenCvSharp, kèm cái bẫy của mỗi bước:

using System.Runtime.InteropServices;
using OpenCvSharp;

/// <summary>Tiền xử lý giống hệt tien_xu_ly() của Python. Ghi kết quả CHW (R, G, B) vào buffer.</summary>
sealed class TienXuLy
{
    private readonly int _size;
    private readonly float[] _meanRgb, _stdRgb;

    public TienXuLy(int size, float[] meanRgb, float[] stdRgb)
    {
        _size = size; _meanRgb = meanRgb; _stdRgb = stdRgb;
    }

    public int SoPhanTu => 3 * _size * _size;

    public void Chay(Mat bgr, float[] buffer, int offset = 0)
    {
        if (bgr.Empty() || bgr.Type() != MatType.CV_8UC3)
            throw new ArgumentException("Cần ảnh BGR 8 bit 3 kênh");

        using var nho = new Mat();
        Cv2.Resize(bgr, nho, new Size(_size, _size), 0, 0, InterpolationFlags.Area);   // bước 1
        using var rgb = new Mat();
        Cv2.CvtColor(nho, rgb, ColorConversionCodes.BGR2RGB);                          // bước 2

        Mat[] kenh = Cv2.Split(rgb);            // 3 Mat CV_8UC1 theo thứ tự R, G, B
        try
        {
            int n = _size * _size;
            using var f = new Mat();
            for (int c = 0; c < 3; c++)
            {
                double alpha = 1.0 / (255.0 * _stdRgb[c]);                             // bước 3 + 4
                double beta = -_meanRgb[c] / _stdRgb[c];
                kenh[c].ConvertTo(f, MatType.CV_32FC1, alpha, beta);
                Marshal.Copy(f.Data, buffer, offset + c * n, n);                       // bước 5: kênh c liền một khối
            }
        }
        finally
        {
            foreach (var k in kenh) k.Dispose();
        }
    }
}

Vài chi tiết C# đáng chú ý: mọi Mat trung gian đều nằm trong using (mỗi Mat giữ bộ nhớ gốc; quên Dispose trong vòng lặp 30 ảnh/giây là bộ nhớ tăng dần tới khi ứng dụng sập sau vài giờ); Marshal.Copy chép thẳng từ con trỏ dữ liệu của Mat, an toàn vì Mat mới tạo bởi ConvertTo luôn liên tục trong bộ nhớ; tham số offset cho phép ghi nhiều ảnh nối tiếp vào một buffer khi chạy batch.

Đừng tin bảng trên chỉ vì nó "nhìn đúng". Đoạn Python sau mô phỏng y hệt các phép của lớp C# (tách kênh, alpha/beta tính bằng số thực 64 bit như ConvertTo, chép từng kênh liền khối) và so với tien_xu_ly:

6. Chạy suy luận: DenseTensor, NamedOnnxValue, đo thời gian

Đường gọi lâu đời và phổ biến nhất của ONNX Runtime C#: bọc mảng float[] thành DenseTensor<float> có shape NCHW, đặt tên bằng NamedOnnxValue.CreateFromTensor, gọi session.Run. 8. SessionOptions và Execution Provider

Execution Provider (EP) là "phần phụ trợ phần cứng" mà ONNX Runtime giao việc tính toán: CPU, CUDA, TensorRT, DirectML, OpenVINO… Bạn đăng ký EP trong SessionOptions theo thứ tự ưu tiên; phép toán nào EP không hỗ trợ sẽ rơi về CPU.

using Microsoft.ML.OnnxRuntime;

static InferenceSession TaoSession(string modelPath, string thietBi)
{
    using var opt = new SessionOptions
    {
        GraphOptimizationLevel = GraphOptimizationLevel.ORT_ENABLE_ALL,
        IntraOpNumThreads = Math.Max(1, Environment.ProcessorCount / 2),   // chừa CPU cho camera, PLC, giao diện
        LogSeverityLevel = OrtLoggingLevel.ORT_LOGGING_LEVEL_WARNING
    };
    switch (thietBi)
    {
        case "cuda": opt.AppendExecutionProvider_CUDA(0); break;    // cần gói Microsoft.ML.OnnxRuntime.Gpu
        case "dml":  opt.AppendExecutionProvider_DML(0);  break;    // cần gói Microsoft.ML.OnnxRuntime.DirectML
        default: break;                                             // chỉ CPU
    }
    return new InferenceSession(modelPath, opt);   // các tùy chọn đã được chép vào session
}

9. Giải mã đầu ra phát hiện vật: letterbox, NMS, đổi tọa độ

Mô hình phát hiện vật (YOLO và tương tự) cần hai việc mà phân loại không có. Letterbox: thu nhỏ ảnh giữ nguyên tỉ lệ rồi đệm viền cho đủ hình vuông (ví dụ 640×640), để vật không bị méo. Sau suy luận phải làm ngược lại: trừ phần đệm, chia tỉ lệ, để tọa độ khung về ảnh gốc. NMS (Non-Maximum Suppression, loại bỏ khung trùng): mô hình đề xuất nhiều khung chồng lên cùng một vật, chỉ giữ khung điểm cao nhất trong mỗi nhóm có IoU lớn.

import numpy as np


def letterbox_thong_so(w, h, dich=640):
    r = min(dich / w, dich / h)
    nw, nh = round(w * r), round(h * r)
    return r, (dich - nw) // 2, (dich - nh) // 2                 # tỉ lệ, đệm trái, đệm trên


def iou(a, b):
    rong = max(0.0, min(a[2], b[2]) - max(a[0], b[0]))
    cao = max(0.0, min(a[3], b[3]) - max(a[1], b[1]))
    giao = rong * cao
    return giao / ((a[2] - a[0]) * (a[3] - a[1]) + (b[2] - b[0]) * (b[3] - b[1]) - giao)


def giai_ma_tho(out, r, pad_x, pad_y, conf=0.25, iou_max=0.45):
    """out: (1, 4 + nc, N). Trả về [(x1, y1, x2, y2, diem, lop)] theo tọa độ ảnh gốc."""
    p = out[0]
    diem_lop = p[4:]                                               # (nc, N)
    lop = diem_lop.argmax(0)
    diem = diem_lop.max(0)
    ung_vien = []
    for i in np.flatnonzero(diem >= conf):
        cx, cy, w, h = p[:4, i]
        ung_vien.append([cx - w / 2, cy - h / 2, cx + w / 2, cy + h / 2, float(diem[i]), int(lop[i])])
    ung_vien.sort(key=lambda k: -k[4])
    giu = []
    for k in ung_vien:                                             # NMS theo từng lớp
        if all(k[5] != g[5] or iou(k, g) < iou_max for g in giu):
            giu.append(k)
    print(f"đề xuất {p.shape[1]}, qua ngưỡng điểm {len(ung_vien)}, sau NMS {len(giu)}")
    return [((x1 - pad_x) / r, (y1 - pad_y) / r, (x2 - pad_x) / r, (y2 - pad_y) / r, d, c)
            for x1, y1, x2, y2, d, c in giu]

Hai khung đúng vị trí ban đầu (sai khác dưới 0,1 pixel do làm tròn float32), khung trùng của vật A bị NMS loại, khung điểm 0,20 bị ngưỡng loại. Bản C# tương đương, làm việc trực tiếp trên mảng phẳng (truy cập Tensor theo chỉ số nhiều chiều chậm hơn nhiều với 8400 cột):

record struct Khung(float X1, float Y1, float X2, float Y2, float Diem, int Lop);

static float IoU(Khung a, Khung b)
{
    float w = MathF.Max(0, MathF.Min(a.X2, b.X2) - MathF.Max(a.X1, b.X1));
    float h = MathF.Max(0, MathF.Min(a.Y2, b.Y2) - MathF.Max(a.Y1, b.Y1));
    float giao = w * h;
    return giao / ((a.X2 - a.X1) * (a.Y2 - a.Y1) + (b.X2 - b.X1) * (b.Y2 - b.Y1) - giao);
}

/// out: mảng phẳng của tensor [1, 4 + nc, n]; phần tử (kênh k, đề xuất i) nằm ở out[k * n + i]
static List<Khung> GiaiMaTho(float[] o, int nc, int n, float r, int padX, int padY,
                             float conf = 0.25f, float iouMax = 0.45f)
{
    var ungVien = new List<Khung>();
    for (int i = 0; i < n; i++)
    {
        int lop = 0; float diem = o[4 * n + i];
        for (int c = 1; c < nc; c++)
            if (o[(4 + c) * n + i] > diem) { diem = o[(4 + c) * n + i]; lop = c; }
        if (diem < conf) continue;
        float cx = o[i], cy = o[n + i], w = o[2 * n + i], h = o[3 * n + i];
        ungVien.Add(new Khung(cx - w / 2, cy - h / 2, cx + w / 2, cy + h / 2, diem, lop));
    }
    ungVien.Sort((a, b) => b.Diem.CompareTo(a.Diem));
    var giu = new List<Khung>();
    foreach (var k in ungVien)
        if (giu.All(g => g.Lop != k.Lop || IoU(k, g) < iouMax)) giu.Add(k);
    return giu.Select(k => k with
    {
        X1 = (k.X1 - padX) / r, Y1 = (k.Y1 - padY) / r,
        X2 = (k.X2 - padX) / r, Y2 = (k.Y2 - padY) / r
    }).ToList();
}

// Letterbox phía C#: phải giống phía huấn luyện (màu đệm, cách làm tròn, nội suy)
static (Mat anh, float r, int padX, int padY) Letterbox(Mat bgr, int dich = 640)
{
    float r = MathF.Min((float)dich / bgr.Width, (float)dich / bgr.Height);
    int nw = (int)MathF.Round(bgr.Width * r), nh = (int)MathF.Round(bgr.Height * r);
    int padX = (dich - nw) / 2, padY = (dich - nh) / 2;
    using var nho = new Mat();
    Cv2.Resize(bgr, nho, new Size(nw, nh), 0, 0, InterpolationFlags.Linear);
    var ra = new Mat();
    Cv2.CopyMakeBorder(nho, ra, padY, dich - nh - padY, padX, dich - nw - padX,
                       BorderTypes.Constant, new Scalar(114, 114, 114));
    return (ra, r, padX, padY);
}
Letterbox của thư viện huấn luyện có thể khác của bạnUltralytics đệm màu xám 114 và làm tròn phần đệm theo cách riêng; các thư viện khác có thể đặt ảnh ở góc trên trái thay vì giữa. Lệch 1 pixel đệm làm mọi khung lệch 1 pixel, đủ để một phép đo vị trí bị sai. Cách chắc chắn: đưa vài ảnh phát hiện vật vào bộ ảnh vàng, ghi khung mong đợi, và so khung bằng ghép cặp IoU như bài trước.

def bien_the(anh_bgr, loi):
    noi_suy = cv2.INTER_LINEAR if loi == "resize Linear" else cv2.INTER_AREA
    a = cv2.resize(anh_bgr, (KICH_THUOC, KICH_THUOC), interpolation=noi_suy)

    a = a.astype(np.float32)
    if loi != "quên chia 255":
        a = a / 255.0
    m, s = (MEAN[::-1], STD[::-1]) if loi == "mean/std B-G-R" else (MEAN, STD)
    return np.ascontiguousarray(((a - m) / s).transpose(2, 0, 1), dtype=np.float32)

Tóm tắt

  • Gói Microsoft.ML.OnnxRuntime cho CPU; GPU NVIDIA dùng Microsoft.ML.OnnxRuntime.Gpu (thay thế, không cài cùng); Windows mọi GPU dùng .DirectML; Intel dùng gói OpenVINO EP.
  • InferenceSession tạo một lần, dùng chung (Run an toàn đa luồng); SessionOptions đặt mức tối ưu đồ thị, số luồng, EP. Mọi đối tượng giữ bộ nhớ gốc (session, options, kết quả Run, Mat) phải nằm trong using.
  • Tiền xử lý C# phải giống hệt Python: nội suy Area, BGR→RGB, ConvertTo(alpha = 1/(255·std), beta = −mean/std) từng kênh, chép từng kênh liền khối để có NCHW.
  • DenseTensor<float> bọc một float[] dùng lại cho mọi ảnh; NamedOnnxValue.CreateFromTensor + session.Run; chép đầu ra ra mảng trước khi Dispose. API OrtValue là lựa chọn mới hơn.
  • Phân loại: softmax/sigmoid rồi ngưỡng. Phát hiện vật: in shape đầu ra, giải mã đúng dạng (thô cần NMS, end-to-end không cần), bỏ letterbox để về tọa độ ảnh gốc.
  • Parity test với bộ ảnh vàng (mã băm mô hình, logit trong dung sai, tổng từng kênh khi lệch) là điều kiện để lắp chương trình lên trạm.
